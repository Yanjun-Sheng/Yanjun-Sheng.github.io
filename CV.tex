%-------------------- RESEARCH EXPERIENCE --------------------%
\newpage
\cvsection{Research Experience}

\cventry
  {PhD Researcher}
  {Aug 2023 -- Present}
  {Australian National University, RSAA}
  {Canberra, Australia}

\begin{itemize}
    \item Built and ran the HaloDance suite of $\sim$3,000 high-resolution N-body simulations to map the dynamical response of the Milky Way stellar halo to the LMC.
    \item Designed robust statistical summaries of velocity moments, anisotropy, and spatial structure, enabling direct comparisons between simulations and stellar-halo observations.
    \item Developed a production simulation-based inference pipeline combining field-level and summary-level approaches to constrain Milky Way--LMC parameters from phase-space data.
    \item Currently applying the inference pipeline to H3 and SDSS-V stellar surveys to obtain observational measurements of the Milky Way--LMC system.
\end{itemize}
